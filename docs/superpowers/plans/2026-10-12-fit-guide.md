# 분량 맞추기(pages)와 작성 요령 체크리스트(guide) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 한글로 장별 쪽수를 재는 `pages`, 양식의 작성 요령·안내·배점을 장별로 뽑는 `guide`, 그리고 Claude가 그 결과로 분량을 맞추고 체크리스트를 만드는 스킬 안내를 만든다.

**Architecture:** `hwpxkit/fit.py`가 분량 제한·배점·장 범위 계산과 작성 요령 뽑기를 맡는다(순수 파이썬, 시험 쉬움). 쪽 번호는 `bridge_hwp.ps1`의 새 동작 `pagemap`이 숨은 한글에서 문단마다 `KeyIndicator`로 얻고, `bridge.page_map`이 돌려준다. CLI `pages`·`guide`가 둘을 잇는다.

**Tech Stack:** Python 3.10+, lxml, 한글 COM(PowerShell + C# 리플렉션).

**Spec:** docs/superpowers/specs/2026-10-12-fit-guide-design.md

## Global Constraints

- 서식(글꼴·줄 간격·여백)은 바꾸지 않는다. 분량은 글로 맞춘다.
- `pages`·`guide`는 문서를 바꾸지 않는다(한글에서도 저장하지 않음).
- 원문에 없는 수치·사실을 지어내지 않는다(스킬 안내).
- 새 의존성 없음. 사용자에게 보이는 글은 한국어 존댓말.
- 사용자 실제 문서(private/)는 시험·공개 파일에 쓰지 않는다.

## Review Focus

1. "제3장 이내 작성"·"제2장" 같은 장 번호 — 분량 제한으로 잡히지 않아야 한다 (Task 1 `test_find_limits`).

"""분량(쪽수·제한·배점)과 작성 요령 항목 (설계 3·4장). 판단과 고쳐 쓰기는 Claude가 한다."""
from __future__ import annotations

import re

from .body import own_text
from .header import Header
from .ns import q
from .package import Package
from .samples import bullet_chars, classify, heading_levels

_LIMIT = re.compile(r"(?<![제\d])(\d{1,3})\s*(쪽|페이지|매|장)\s*(이내|이하|내외|까지|以內)")
_SCORE = re.compile(r"[〔(\[]\s*(\d{1,3})\s*점\s*[〕)\]]\s*$")
_HEAD = {"h1": 1, "h2": 2, "h3": 3, "h4": 4, "h5": 5}


def find_limits(texts: list[str]) -> list[dict]:
    out, seen = [], set()
    for text in texts:
        for m in _LIMIT.finditer(text):
            key = m.group(0)
            if key not in seen:
                seen.add(key)
                out.append({"pages": int(m.group(1)), "kind": m.group(3), "text": key})
    return out


def heading_score(text: str) -> int | None:
    m = _SCORE.search(text.strip())
    return int(m.group(1)) if m else None


def _tops(pkg: Package) -> list:
    return [p for p in pkg.xml(pkg.section_names()[0]) if p.tag == q("hp:p")]


def headings(pkg: Package, level: int = 2) -> list[tuple[int, str, int]]:
    h = Header(pkg)
    chars = bullet_chars(h)
    tops = _tops(pkg)
    levels = heading_levels(tops)
    out = []
    for i, p in enumerate(tops):
        lvl = _HEAD.get(classify(p, h, chars, levels)[0])
        if lvl is not None and lvl <= level:
            out.append((i, own_text(p).strip(), lvl))
    return out


def sections_from_pages(heads, pages: list[int], total: int) -> list[dict]:
    out = []
    for k, (i, title, lvl) in enumerate(heads):
        if i >= len(pages):
            break
        start = pages[i]
        nxt = next((j for j, _, l2 in heads[k + 1:] if l2 <= lvl and j < len(pages)), None)
        end = max(start, pages[nxt] - 1) if nxt is not None else max(start, total)
        out.append({"title": title, "level": lvl, "score": heading_score(title), "para": i,
                    "start": start, "end": end, "count": end - start + 1})
    return out


def doc_texts(pkg: Package) -> list[str]:
    return [t for sec in pkg.section_names() for p in pkg.xml(sec).iter(q("hp:p"))
            if (t := " ".join(own_text(p).split()))]


def measure(pkg: Package, pages: list[int], total: int, *, level: int = 2, limit: int | None = None) -> dict:
    warnings = []
    n_tops = len(_tops(pkg))
    if len(pages) != n_tops:
        warnings.append(f"한글이 센 문단 수({len(pages)})와 문서 문단 수({n_tops})가 달라 앞쪽부터 맞췄어요. "
                        "장 쪽수가 조금 다를 수 있어요.")
        pages = pages[:n_tops]
    limits = find_limits(doc_texts(pkg))
    if limit is not None:
        chosen = {"pages": limit, "kind": "이내", "text": "--limit"}
    else:
        chosen = limits[0] if len(limits) == 1 else None
    return {"pages": total, "limit": chosen, "limits": limits,
            "sections": sections_from_pages(headings(pkg, level), pages, total), "warnings": warnings}
```

_BOX_HEAD = re.compile(r"작성\s*(요령|방법|안내)|유의\s*사항|기재\s*요령")
_NOTE_WORDS = re.compile(r"작성|기술|기재|포함|제시|첨부|명시")


def _box_lines(tbl) -> list[str] | None:
    """작성 요령 상자(1칸 표, 또는 첫 칸이 '작성 요령' 같은 머리인 표)의 항목 줄. 상자가 아니면 None."""
    tcs = list(tbl.iter(q("hp:tc")))
    first = " ".join(own_text(p) for p in tcs[0].iter(q("hp:p"))).strip() if tcs else ""
    if not (len(tcs) == 1 or _BOX_HEAD.search(first)):
        return None
    out = []
    for p in tbl.iter(q("hp:p")):
        text = " ".join(own_text(p).split())
        if not text or (_BOX_HEAD.search(text) and len(text) <= 12):  # 빈 줄, 머리 줄('작성 요령')
            continue
        out.append(text)
    return out


def guide_items(pkg: Package) -> dict:
    """양식의 작성 요령·안내·배점을 장(h1~h2)별로."""
    from .review import _GUIDE_RE
    tops = _tops(pkg)
    head_at = {i: (title, lvl) for i, title, lvl in headings(pkg, 2)}
    sections = [{"title": "(앞부분)", "score": None, "items": []}]
    seen: set[str] = set()

    def add(kind: str, text: str) -> None:
        if text not in seen:
            seen.add(text)
            sections[-1]["items"].append({"kind": kind, "text": text})

    for i, p in enumerate(tops):
        if i in head_at:
            title = head_at[i][0]
            score = heading_score(title)
            sections.append({"title": title, "score": score, "items": []})
            if score is not None:
                sections[-1]["items"].append({"kind": "배점", "text": f"{score}점"})
            continue
        tbl = next((t for run in p.findall(q("hp:run")) for t in run.findall(q("hp:tbl"))), None)
        if tbl is not None:
            for line in _box_lines(tbl) or []:
                add("요령", line)
            continue
        text = " ".join(own_text(p).split())
        if text and (_GUIDE_RE.search(text) or (text.startswith("※") and _NOTE_WORDS.search(text))):
            add("안내", text)
    return {"limits": find_limits(doc_texts(pkg)), "sections": [s for s in sections if s["items"]]}
```

"※ 참고: 이 줄은 안내가 아님"은 `_NOTE_WORDS`가 없어 빠진다. 제3장은 항목이 없어 결과에서 빠진다.

def _cmd_pages(args, workdir) -> int:
    from . import fit
    src = _resolve_template(args.file)
    pkg = _open_any(src, workdir)
    if not bridge.available():
        raise PackageError("한글이 있어야 쪽수를 잴 수 있어요. 한글이 설치된 PC에서 다시 해 주세요.")
    total, paras = bridge.page_map(src)
    r = fit.measure(pkg, paras, total, level=args.level, limit=args.limit)
    if args.json:
        print(json.dumps(r, ensure_ascii=False))
        return 0
    lim = r["limit"]
    head = f"전체 {total}쪽"
    if lim:
        over = total - lim["pages"]
        state = f"{over}쪽 넘음" if over > 0 else f"{-over}쪽 남음" if over < 0 else "딱 맞음"
        head += f" (제한 {lim['pages']}쪽 {lim['kind']} → {state})"
    elif r["limits"]:
        head += " (분량 제한이 여러 개예요: " + ", ".join(x["text"] for x in r["limits"]) + ")"
    else:
        head += " (분량 제한을 찾지 못했어요. 제한이 몇 쪽인가요?)"
    print(head)
    for s in r["sections"]:
        indent = "  " * (s["level"] - 1)
        span = f"{s['start']}쪽" if s["start"] == s["end"] else f"{s['start']}~{s['end']}쪽"
        print(f"{indent}{s['title']}  {span} ({s['count']}쪽)")
    for w in r["warnings"]:
        print(f"[주의] {w}")
    return 0

# 분량 맞추기와 작성 요령 체크리스트

## 분량 맞추기 ("10쪽 이내로 줄여줘")

1. `python "${CLAUDE_PLUGIN_ROOT}/hwpx.py" pages <문서>` — 장별 쪽수와 분량 제한. 제한을 못 찾았거나 여러 개면 사용자에게 묻고 `--limit N`.
2. 넘친 쪽수를 장마다 나눈다. 배점이 있으면 "쪽수 ÷ 배점"이 큰 장부터, 없으면 쪽수가 많은 장부터. 장마다 줄일 쪽수를 표로 보여 준다.
3. 장마다 `read --anchors`로 읽고 줄일 곳을 고른다.
   - 줄이는 법: 겹치는 설명 합치기, 군더더기·반복 빼기, 긴 문장 나누기, 개조식 다듬기.
   - 지킬 것: 수치, 고유명사, 배점 항목에 답하는 문장, 표·그림·캡션. 표·그림을 줄여야 하면 따로 묻는다.
   - "빼는 문장 / 합치는 문장"을 보여 주고 확인받는다.
4. 확인받은 장만 `render <문서> <새.md> <새결과.hwpx> --replace a:b`로 바꾼다. 원본은 덮어쓰지 않는다.
5. 다시 `pages`. 제한 안이면 끝. 최대 3번까지 되풀이하고, 그래도 넘치면 남은 쪽수와 다음 후보를 보여 주고 멈춘다.
6. 모자라면 배점 큰 장부터 보충한다. 원문·사용자 자료에 없는 수치·사실은 지어내지 않고 필요한 자료를 묻는다.

## 작성 요령 체크리스트 ("양식대로 다 썼는지 봐줘")

1. 원본 양식에 `python "${CLAUDE_PLUGIN_ROOT}/hwpx.py" guide <양식>` — 장별 작성 요령·안내·배점·분량. 원본이 없으면 다 쓴 문서에 실행하고 "남은 안내만 봤어요"라고 알린다.
